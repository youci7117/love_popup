import tkinter as tk
import random
import threading

MAX_WINDOWS = 30       # 最多弹出 30 个窗口
CREATE_INTERVAL = 100  # 间隔 100 毫秒创建一个新窗口

all_windows = []       # 记录所有窗口
exit_flag = False      


def show_love():
    """创建一个告白弹窗"""
    global exit_flag

    if exit_flag:
        return

    try:
        window = tk.Toplevel()
        all_windows.append(window)

        # 屏幕大小
        screen_w = window.winfo_screenwidth()
        screen_h = window.winfo_screenheight()

        # 设置窗口尺寸
        win_w = 260
        win_h = 120

        # 随机位置
        x = random.randint(0, screen_w - win_w)
        y = random.randint(0, screen_h - win_h)

        window.title("给你的小提示")
        window.geometry(f"{win_w}x{win_h}+{x}+{y}")
        window.attributes("-topmost", True)

        # 显示的情话内容
        tips = [
            "我好喜欢你哦 ❤️",
            "你是我每天的心动来源！",
            "让你被我喜欢一下，不介意吧？",
            "今天也要喜欢我一点点噢！",
            "你在我心里占据了柔软的一角。",
            "见到你我就开心！",
            "你知道吗？你真的很可爱。",
            "我好想你(///▽///)",
        ]

        colors = ["pink", "lavenderblush", "lightyellow", "mistyrose", "peachpuff"]

        tk.Label(
            window,
            text=random.choice(tips),
            bg=random.choice(colors),
            font=("Microsoft YaHei", 14),
            wraplength=230
        ).pack(expand=True, fill="both", padx=10, pady=10)

        # 禁用关闭
        window.protocol("WM_DELETE_WINDOW", lambda: None)

    except:
        pass


def auto_create():
    """持续生成弹窗"""
    global exit_flag

    count = 0
    while count < MAX_WINDOWS and not exit_flag:
        show_love()
        count += 1
        root.after(CREATE_INTERVAL, lambda: None)


def force_exit(event=None):
    """关闭所有弹窗"""
    global exit_flag
    exit_flag = True
    for w in all_windows:
        try:
            w.destroy()
        except:
            pass
    try:
        root.destroy()
    except:
        pass


# 主窗口
root = tk.Tk()
root.title("💕 小小心意")
root.geometry("300x120")
root.attributes("-topmost", True)

tk.Label(
    root,
    text="点开我就代表你被喜欢啦！💕",
    font=("Microsoft YaHei", 14)
).pack(expand=True)

# 按快捷键 Ctrl+Q 关闭
root.bind("<Control-q>", force_exit)

# 开始自动弹窗
threading.Thread(target=auto_create).start()

root.mainloop()
